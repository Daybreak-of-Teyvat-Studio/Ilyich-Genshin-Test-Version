# ---------- 阶段 2：弱势收手（打不动就先整补） ----------

	allowed = {
		OR = {
			tag = FOD
			tag = FOM
			tag = HZH
			tag = PBF
		}
	}
	enable = {
		any_enemy_country = {
			OR = {
				tag = FOD

				tag = HZH
				tag = PBF
			}
		}
		has_war = yes
		NOT = { has_global_flag = FON_cw_startup }
		OR = {
			surrender_progress > 0.15

				fighting_army_strength_ratio = {
