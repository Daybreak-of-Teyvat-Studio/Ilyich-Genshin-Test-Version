############################################################
# 璃月内战 · 六方机制挂载（合并文件）
# 钩子：on_startup 初始化内战机制；on_daily 每日保活（战后机制仅在本方完成统一后保活）
############################################################
on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = {
					OR = {
						tag = KQP
						tag = LYY
						tag = GYP
						tag = SHP
						tag = YLH
						tag = CYG
					}
				}
				if = {
					limit = { tag = KQP }
					KQP_Init = yes
				}
				if = {
					limit = { tag = LYY }
					LYY_Init = yes
				}
				if = {
					limit = { tag = GYP }
					GYP_Init = yes
				}
				if = {
					limit = { tag = SHP }
					SHP_Init = yes
				}
				if = {
					limit = { tag = YLH }
					YLH_Init = yes
				}
				if = {
					limit = { tag = CYG }
					CYG_Init = yes
				}
			}
		}
	}
	on_daily = {
		effect = {
			every_country = {
				limit = {
					OR = {
						tag = KQP
						tag = LYY
						tag = GYP
						tag = SHP
						tag = YLH
						tag = CYG
					}
				}
				if = {
					limit = { tag = KQP }
					KQP_Sync = yes
				}
				if = {
					limit = { tag = KQP  has_country_flag = KQP_PostWar_Active }
					KQPPW_Sync = yes
				}
				if = {
					limit = { tag = LYY }
					LYY_Sync = yes
				}
				if = {
					limit = { tag = LYY  has_country_flag = LYY_PostWar_Active }
					LYYPW_Sync = yes
				}
				if = {
					limit = { tag = GYP }
					GYP_Sync = yes
				}
				if = {
					limit = { tag = GYP  has_country_flag = GYP_PostWar_Active }
					GYPPW_Sync = yes
				}
				if = {
					limit = { tag = SHP }
					SHP_Sync = yes
				}
				if = {
					limit = { tag = SHP  has_country_flag = SHP_PostWar_Active }
					SHPPW_Sync = yes
				}
				if = {
					limit = { tag = YLH }
					YLH_Sync = yes
				}
				if = {
					limit = { tag = YLH  has_country_flag = YLH_PostWar_Active }
					YLHPW_Sync = yes
				}
				if = {
					limit = { tag = CYG }
					CYG_Sync = yes
				}
				if = {
					limit = { tag = CYG  has_country_flag = CYG_PostWar_Active }
					CYGPW_Sync = yes
				}
			}
		}
	}
}
