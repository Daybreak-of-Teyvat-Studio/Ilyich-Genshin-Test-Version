KQP_Tier_Refresh = {
	remove_dynamic_modifier = { modifier = KQP_Reform_Tier_1 }
	remove_dynamic_modifier = { modifier = KQP_Reform_Tier_2 }
	remove_dynamic_modifier = { modifier = KQP_Reform_Tier_3 }
	remove_dynamic_modifier = { modifier = KQP_Reform_Tier_4 }
	remove_dynamic_modifier = { modifier = KQP_Reform_Tier_5 }

	remove_dynamic_modifier = { modifier = KQP_Combo_A_High }
	remove_dynamic_modifier = { modifier = KQP_Combo_B_High }
	if = {
		limit = {
			has_variable = KQP-Reform-Index
			NOT = { check_variable = { KQP-Reform-Index > 20 } }
		}
		add_dynamic_modifier = { modifier = KQP_Reform_Tier_1 }
	}
	if = {
		limit = {
			has_variable = KQP-Reform-Index
			check_variable = { KQP-Reform-Index > 20 }
			NOT = { check_variable = { KQP-Reform-Index > 40 } }
		}
		add_dynamic_modifier = { modifier = KQP_Reform_Tier_2 }

	if = {
		limit = {
			has_variable = KQP-Reform-Index

		add_dynamic_modifier = { modifier = KQP_Inertia_Tier_2 }
	}
	if = {

			has_variable = KQP-Inertia-Index

			NOT = { check_variable = { KQP-Inertia-Index > 60 } }

			has_variable = KQP-Reform-Index

			check_variable = { KQP-Reform-Index > 60 }
			check_variable = { KQP-Inertia-Index > 60 }
