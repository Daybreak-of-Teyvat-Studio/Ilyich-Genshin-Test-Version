country_event = {
	id = DVA_tuohuang_events.3
	is_triggered_only = yes
	hidden = yes

	immediate = {
		DVA_th_peace_out = yes
		for_each_scope_loop = {
			array = lw_states
			owner = {
				transfer_state_to = ROOT
			}
		}
		add_to_variable = { DVA_tuohuang_wars_won = 1 }
		DVA_th_clear_variables = yes
		DVA_th_re_add_to_old_faction = yes
		if = {
			limit = {
				check_variable = { DVA_tuohuang_wars_won > 0 }
				NOT = { has_global_flag = DVA_signora_arrived }
			}
			country_event = { id = DVA_tuohuang_events.10 hours = 6 }
		}
		if = {
			limit = {
				all_of_scopes = {
					array = DVA_th_states
					is_owned_by = DVA
				}
			}
			set_global_flag = DVA_tuohuang_done
			country_event = { id = DVA_tuohuang_events.11 hours = 6 }
		}
	}
}
