# ============ 深渊侵蚀度 / 火神恩情：当前档位的国家修正 ============
# 触发条件直接读变量（与各档 dynamic_modifier 的 enable 判定一致），
# 因此不依赖任何档位标记，任何改数值的途径都会立刻反映到描述上。

defined_text = {
	name = NAT_abyss_tier_mods
	text = {
		trigger = { check_variable = { NAT-ABY-Index > 100 } }
		localization_key = NAT_abyss_mods_7
	}
	text = {
		trigger = { check_variable = { NAT-ABY-Index > 60 } NOT = { check_variable = { NAT-ABY-Index > 100 } } }
		localization_key = NAT_abyss_mods_6
	}
	text = {
		trigger = { check_variable = { NAT-ABY-Index > 30 } NOT = { check_variable = { NAT-ABY-Index > 60 } } }
		localization_key = NAT_abyss_mods_5
	}
	text = {
		trigger = { check_variable = { NAT-ABY-Index > 0 } NOT = { check_variable = { NAT-ABY-Index > 30 } } }
		localization_key = NAT_abyss_mods_4
	}
	text = {
		trigger = { check_variable = { NAT-ABY-Index > -20 } NOT = { check_variable = { NAT-ABY-Index > 0 } } }
		localization_key = NAT_abyss_mods_3
	}
	text = {
		trigger = { check_variable = { NAT-ABY-Index > -60 } NOT = { check_variable = { NAT-ABY-Index > -20 } } }
		localization_key = NAT_abyss_mods_2
	}
	text = {
		trigger = { has_variable = NAT-ABY-Index }
		localization_key = NAT_abyss_mods_1
	}
	text = {
		localization_key = NAT_abyss_mods_none
	}
}
