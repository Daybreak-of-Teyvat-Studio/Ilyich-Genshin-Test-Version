# 机制初始化：只置初值，不预挂任何档位
SUM_Neo_Init = {
	hidden_effect = {
	if = {
		limit = { NOT = { has_variable = SUM-Oracle-Index } }
		set_variable = { SUM-Oracle-Index = 20 }
		set_variable = { SUM-Canon-Index = 70 }
		set_variable = { SUM-Genesis-Index = 10 }
	}
	clamp_variable = { var = SUM-Oracle-Index min = 0 max = 100 }
	clamp_variable = { var = SUM-Canon-Index min = 0 max = 100 }
	clamp_variable = { var = SUM-Genesis-Index min = 0 max = 100 }
	clamp_variable = { var = SUM-Genesis-Index min = 0 max = 200 }
	SUM_Neo_Tier_Refresh = yes
	}
}

SUM_Neo_A_Up = { add_to_variable = { SUM-Oracle-Index = 5 }  SUM_Neo_Tier_Refresh = yes }
SUM_Neo_B_Up = { add_to_variable = { SUM-Canon-Index = 5 }  SUM_Neo_Tier_Refresh = yes }

# ==================== 档位重整 ====================
# 每一档修正只在推进该轴的国策完成时被挂上，旧档同时卸下。

	hidden_effect = {
		SUM_Neo_RF_Oracle = yes
		SUM_Neo_RF_Canon = yes
		SUM_Neo_RF_Combo = yes
	}
}

# 月度保活（变量丢失补初值；档位丢失重整）
SUM_Neo_Sync = {
	hidden_effect = {
	if = {
		limit = { NOT = { has_variable = SUM-Oracle-Index } }
		SUM_Neo_Init = yes
	}
	SUM_Neo_Tier_Refresh = yes
	}
}

	if = {
		limit = {
			has_variable = SUM-Canon-Index
			has_variable = SUM-Oracle-Index
		}
		if = {
			limit = { check_variable = { SUM-Oracle-Index > 60 } check_variable = { SUM-Canon-Index > 60 } }
			if = {
				limit = { NOT = { has_country_flag = SUM_Neo_Combo_Twin_High_At } }
				SUM_Neo_Combo_Twin_High_Set = yes
			}
		}
		else_if = {
			limit = { check_variable = { SUM-Oracle-Index < 40 } check_variable = { SUM-Canon-Index < 40 } }
			if = {
				limit = { NOT = { has_country_flag = SUM_Neo_Combo_Twin_Low_At } }
				SUM_Neo_Combo_Twin_Low_Set = yes
			}
		}
		else_if = {
			limit = { check_variable = { SUM-Oracle-Index > 60 } check_variable = { SUM-Canon-Index < 40 } }
			if = {
				limit = { NOT = { has_country_flag = SUM_Neo_Combo_A_High_At } }
				SUM_Neo_Combo_A_High_Set = yes
			}
		}
		else_if = {
			limit = { check_variable = { SUM-Oracle-Index < 40 } check_variable = { SUM-Canon-Index > 60 } }
			if = {
				limit = { NOT = { has_country_flag = SUM_Neo_Combo_B_High_At } }
				SUM_Neo_Combo_B_High_Set = yes
			}
		}
		else = {
			if = {
				limit = {
					OR = {
						has_country_flag = SUM_Neo_Combo_Twin_High_At
						has_country_flag = SUM_Neo_Combo_Twin_Low_At
						has_country_flag = SUM_Neo_Combo_A_High_At
						has_country_flag = SUM_Neo_Combo_B_High_At
					}
				}
				SUM_Neo_None = yes
			}
		}
	}
}
