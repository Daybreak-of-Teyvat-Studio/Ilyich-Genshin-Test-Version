	on_capitulation = {
		effect = {
			#满足下面任一limit里面的条件才会刷战争结语
			#璃月
			IF = {
				limit = {
					#只要败者是璃月势力就触发
					FROM = {
						Is_LYY = yes
					}
				}
				#触发窗口

				FROM = { save_global_event_target_as = winning_country }
