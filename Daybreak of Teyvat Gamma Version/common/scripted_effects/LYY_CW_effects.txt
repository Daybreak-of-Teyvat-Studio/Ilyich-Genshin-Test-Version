# 档位重整：先卸下全部档位，再按当前轴值挂上唯一对应的一档；
# 每一档只在推进该轴的国策完成时被挂上（按国策分阶段赐予）。
KQP_Tier_Refresh = {
	hidden_effect = {
		KQP_RF_Reform = yes
		KQP_RF_Inertia = yes
		KQP_RF_Combo = yes
	}
}

# 月度保活（on_monthly 调用）
KQP_Sync = {
	hidden_effect = {
		if = {
			limit = { NOT = { has_variable = KQP-Reform-Index } }
			KQP_Init = yes
		}
		KQP_Tier_Refresh = yes
	}
}

# CYGPW 分档刷新（档位未变则不做任何事）
##########################

CYGPW_RF_PWSoil = {
	if = {
		limit = {
			has_variable = CYG-PWSoil-Index
		}
		if = {
			limit = { NOT = { check_variable = { CYG-PWSoil-Index > 20 } } }
			if = {
				limit = { NOT = { has_country_flag = CYG_PW_Soil_Tier_1_At } }
				CYG_PW_Soil_Tier_1_Set = yes
			}
		}
		else_if = {
			limit = { check_variable = { CYG-PWSoil-Index > 20 } NOT = { check_variable = { CYG-PWSoil-Index > 40 } } }
			if = {
				limit = { NOT = { has_country_flag = CYG_PW_Soil_Tier_2_At } }
				CYG_PW_Soil_Tier_2_Set = yes
			}
		}
		else_if = {
			limit = { check_variable = { CYG-PWSoil-Index > 40 } NOT = { check_variable = { CYG-PWSoil-Index > 60 } } }
			if = {
				limit = { NOT = { has_country_flag = CYG_PW_Soil_Tier_3_At } }
				CYG_PW_Soil_Tier_3_Set = yes
			}
		}
		else_if = {
			limit = { check_variable = { CYG-PWSoil-Index > 60 } NOT = { check_variable = { CYG-PWSoil-Index > 80 } } }
			if = {
				limit = { NOT = { has_country_flag = CYG_PW_Soil_Tier_4_At } }
				CYG_PW_Soil_Tier_4_Set = yes
			}
		}
		else_if = {
			limit = { check_variable = { CYG-PWSoil-Index > 80 } }
			if = {
				limit = { NOT = { has_country_flag = CYG_PW_Soil_Tier_5_At } }
				CYG_PW_Soil_Tier_5_Set = yes
			}
		}
	}
}
