# 四、月度结算（深渊侵蚀度）
#       挂载点：common/on_actions/DOT_NAT_on_actions.txt 的 on_monthly_NAT
# ==============================================================

NAT_ABY_Monthly_Tick = {
	# —— 4.1 自然漂移（档位越高涨得越快）——
	if = {
		limit = { check_variable = { NAT-ABY-Index > 100 } }
		add_to_variable = { var = NAT-ABY-Index value = 5 }
	}
	else_if = {
		limit = { check_variable = { NAT-ABY-Index > 60 } }
		add_to_variable = { var = NAT-ABY-Index value = 3 }
	}
	else_if = {
		limit = { check_variable = { NAT-ABY-Index > 30 } }
		add_to_variable = { var = NAT-ABY-Index value = 2 }
	}
	else_if = {
		limit = { check_variable = { NAT-ABY-Index > 0 } }
		add_to_variable = { var = NAT-ABY-Index value = 1 }
	}
	else_if = {
		limit = { check_variable = { NAT-ABY-Index > -60 } }
		add_to_variable = { var = NAT-ABY-Index value = -1 }
	}
	else = {
		add_to_variable = { var = NAT-ABY-Index value = -3 }
	}

	# —— 4.2 路线丙「秘源核心」：科研快，但深渊在啃噬 ——
	if = {
		limit = { has_idea = NAT_Secret_Source_Core }
		add_to_variable = { var = NAT-ABY-Index value = 1 }
	}

	# —— 4.3 古龙系古名：每 3 个古名，地脉每 90 天自我修复 3 点 ——
	if = {
		limit = {
			check_variable = { NAT-Anance-Index > 5 }
			NOT = { has_country_flag = NAT_Anance_Heal_CD }
		}
		set_country_flag = { flag = NAT_Anance_Heal_CD days = 90 }
		add_to_variable = { var = NAT-ABY-Index value = -3 }
	}

	# —— 4.4 燃素天网：90 天内侵蚀度不再自然上涨（用 flag 表示窗口期）——
	if = {
		limit = { has_country_flag = NAT_Ember_Net_Window }
		add_to_variable = { var = NAT-ABY-Index value = -4 }
	}

	# —— 4.5 深渊之潮：侵蚀度 > 30 时滚动触发（冷却 180 天）——
	if = {
		limit = {
			check_variable = { NAT-ABY-Index > 30 }
			NOT = { has_country_flag = NAT_Abyss_Tide_CD }
		}
		set_country_flag = { flag = NAT_Abyss_Tide_CD days = 180 }
		random_list = {
			45 = {
				country_event = { id = NAT_Event.500 days = 2 random_days = 8 }
			}
			55 = {
				country_event = { id = NAT_Event.501 days = 2 random_days = 8 }
			}
		}
	}

	# —— 4.6 七级深渊化 —— 深渊之蚀宣战（只触发一次）——
	if = {
		limit = {
			check_variable = { NAT-ABY-Index > 140 }
			NOT = { has_country_flag = NAT_Abyss_War_Triggered }
		}
		NAT_Abyss_War_Trigger = yes
	}

	NAT_Abyss_Sync = yes
}

# 深渊之蚀宣战（统一入口：决议「深渊侵蚀中」主动引爆 / 侵蚀度 > 140 自动引爆）
NAT_Abyss_War_Trigger = {
	set_country_flag = NAT_Abyss_War_Triggered
	every_owned_state = {
		limit = { is_core_of = NAT }
		add_resistance = 0.10
	}
	# 直接宣战，不依赖玩家点击事件选项（事件 210 仅作叙事弹窗）
	NAT_Declare_Abyss_War = yes
	country_event = { id = NAT_Event.210 days = 1 }
	NAT_Abyss_Sync = yes
}

NAT_GRACE_Monthly_Tick = {
	# —— 5.1 自然漂移 ——
	if = {
		limit = { check_variable = { MVK-Prestige-Index > 109 } }
		add_to_variable = { var = MVK-Prestige-Index value = 1 }
	}
	else_if = {
		limit = { check_variable = { MVK-Prestige-Index > 79 } }
		add_to_variable = { var = MVK-Prestige-Index value = 2 }
	}
	else_if = {
		limit = { check_variable = { MVK-Prestige-Index > 59 } }
		add_to_variable = { var = MVK-Prestige-Index value = 2 }
	}
	else_if = {
		limit = { check_variable = { MVK-Prestige-Index > 44 } }
		add_to_variable = { var = MVK-Prestige-Index value = 1 }
	}
	else_if = {
		limit = { check_variable = { MVK-Prestige-Index > 29 } }
		add_to_variable = { var = MVK-Prestige-Index value = 0 }
	}
	else_if = {
		limit = { check_variable = { MVK-Prestige-Index > 14 } }
		add_to_variable = { var = MVK-Prestige-Index value = -1 }
	}
	else = {
		add_to_variable = { var = MVK-Prestige-Index value = -2 }
	}

	# —— 5.2 部族离心（恩情 < 30）：随机一族拒绝输诚 ——
	if = {
		limit = {
			check_variable = { MVK-Prestige-Index < 30 }
			NOT = { has_country_flag = NAT_Tribe_Strain_CD }
		}
		set_country_flag = { flag = NAT_Tribe_Strain_CD days = 90 }
		random_list = {
			35 = { country_event = { id = NAT_Event.150 days = 3 random_days = 10 } }
			65 = { }
		}
	}

	# —— 5.3 部族叛乱（恩情 < 15）——
	if = {
		limit = {
			check_variable = { MVK-Prestige-Index < 15 }
			has_country_flag = NAT_Tribe_Strain_Active
			NOT = { has_country_flag = NAT_Tribe_Revolt_CD }
		}
		set_country_flag = { flag = NAT_Tribe_Revolt_CD days = 120 }
		random_list = {
			45 = { country_event = { id = NAT_Event.160 days = 3 random_days = 10 } }
			55 = { }
		}
	}

	# —— 5.4 六盟崩解宣战（恩情 <= 0 且已有部族在闹）——
	if = {
		limit = {
			check_variable = { MVK-Prestige-Index < 1 }
			has_country_flag = NAT_Tribe_Strain_Active
			NOT = { has_country_flag = NAT_Covenant_Collapse_Done }
		}
		NAT_Covenant_Collapse = yes
	}

	NAT_Grace_Sync = yes
}

# 六盟崩解：全部脱离的部族组成联军向纳塔宣战

	set_country_flag = NAT_Covenant_Collapse_Done
	country_event = { id = NAT_Event.170 days = 1 }
}

	}
}

# 深渊战线：侵蚀度越高，深渊推进越快；侵蚀度被压住，深渊被反推

	if = {
		limit = { check_variable = { NAT-ABY-Index > 79 } }
		NAT_War_Advance_4 = yes
		add_stability = -0.01
		every_owned_state = {
			limit = { is_core_of = NAT }
			add_resistance = 0.01
		}
	}
	else_if = {
		limit = { check_variable = { NAT-ABY-Index > 59 } }
		NAT_War_Advance_3 = yes
		add_stability = -0.005
	}
	else_if = {
		limit = { check_variable = { NAT-ABY-Index > 39 } }
		NAT_War_Advance_2 = yes
	}
	else_if = {
		limit = { check_variable = { NAT-ABY-Index > 19 } }
		NAT_War_Advance_1 = yes
	}
	else = {
		# 侵蚀度被压到极低 → 反推
		NAT_War_Retreat_3 = yes
		NAT_Abyss_Sub_2 = yes
		add_war_support = 0.01
	}
}

	# 关键：打上「已吞并」旗标 —— 触发深渊之蚀进攻的决议靠它失效
	set_country_flag = NAT_Abyss_HIP_Absorbed
	if = {
		limit = { HIP = { exists = yes } }
		# 吞并深渊之蚀（收编其部队）
		annex_country = { target = HIP transfer_troops = yes }
	}
	add_political_power = 25
	add_manpower = 100000
}

# 吞并六大部族之后：对所有邻国宣战。
NAT_Abyss_Natlan_Declare_Neighbor_Wars = {
	# 深渊纳塔的第一场对外战争：四境征伐
	every_neighbor_country = {
		limit = {
			NOT = { tag = NAT }
			NOT = { is_subject_of = NAT }
			NOT = { has_war_with = NAT }
		}
		# 宣战吞并（对邻国一律吞并）
		NAT = { declare_war_on = { target = THIS type = annex_everything } }
	}
	add_war_support = 0.10
	add_stability = 0.05
	set_country_flag = NAT_Abyss_Expansion_Begun
}
