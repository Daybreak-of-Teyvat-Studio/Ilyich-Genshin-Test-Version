############################################################
# 须弥 Neo · 机制挂载
# on_startup 初始化；on_daily 每日保活。
# 注意：内容必须写在 effect = { } 里，裸写 every_country 会被解析器拒绝。
############################################################
on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = { tag = SUM }
				SUM_Neo_Init = yes
			}
			every_country = {
				limit = { tag = SGD }
				SUM_Neo_SGD_Init = yes
			}
			every_country = {
				limit = { tag = SDH }
				SUM_Neo_SDH_Init = yes
			}
			every_country = {
				limit = { tag = SGS }
				SUM_Neo_SGS_Init = yes
			}
		}
	}
	on_daily = {
		effect = {
			every_country = {
				limit = { tag = SUM }
				SUM_Neo_Sync = yes
			}
			every_country = {
				limit = { tag = SGD }
				SUM_Neo_SGD_Sync = yes
			}
			every_country = {
				limit = { tag = SDH }
				SUM_Neo_SDH_Sync = yes
			}
			every_country = {
				limit = { tag = SGS }
				SUM_Neo_SGS_Sync = yes
			}
		}
	}
}
