SAN_PW_Sync = {
	if = {
		limit = { NOT = { has_variable = SAN-Mandate-Index } }
		SAN_PW_Init = yes
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = SAN_PW_Mandate_Tier_1 } } }
		add_dynamic_modifier = { modifier = SAN_PW_Mandate_Tier_1 }
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = SAN_PW_Mandate_Tier_2 } } }
		add_dynamic_modifier = { modifier = SAN_PW_Mandate_Tier_2 }
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = SAN_PW_Mandate_Tier_3 } } }
		add_dynamic_modifier = { modifier = SAN_PW_Mandate_Tier_3 }

	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = SAN_PW_Mandate_Tier_4 } } }
		add_dynamic_modifier = { modifier = SAN_PW_Mandate_Tier_4 }
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = SAN_PW_Mandate_Tier_5 } } }
		add_dynamic_modifier = { modifier = SAN_PW_Mandate_Tier_5 }
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = SAN_PW_Deep_Tier_1 } } }
		add_dynamic_modifier = { modifier = SAN_PW_Deep_Tier_1 }
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = SAN_PW_Deep_Tier_2 } } }
		add_dynamic_modifier = { modifier = SAN_PW_Deep_Tier_2 }
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = SAN_PW_Deep_Tier_3 } } }
		add_dynamic_modifier = { modifier = SAN_PW_Deep_Tier_3 }
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = SAN_PW_Deep_Tier_4 } } }
		add_dynamic_modifier = { modifier = SAN_PW_Deep_Tier_4 }
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = SAN_PW_Deep_Tier_5 } } }
		add_dynamic_modifier = { modifier = SAN_PW_Deep_Tier_5 }
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = SAN_PW_Combo_Twin_High } } }
		add_dynamic_modifier = { modifier = SAN_PW_Combo_Twin_High }
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = SAN_PW_Combo_A_High } } }
		add_dynamic_modifier = { modifier = SAN_PW_Combo_A_High }
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = SAN_PW_Combo_B_High } } }
		add_dynamic_modifier = { modifier = SAN_PW_Combo_B_High }
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = SAN_PW_Combo_Twin_Low } } }
		add_dynamic_modifier = { modifier = SAN_PW_Combo_Twin_Low }
	}
}
