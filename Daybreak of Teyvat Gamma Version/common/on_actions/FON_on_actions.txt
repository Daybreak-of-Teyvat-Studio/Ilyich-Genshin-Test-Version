on_actions = {
    on_capitulation_immediate = {
        effect = {
            # 内战结束：清掉 AI 厌战 idea 与后方登陆旗（输家被吞并，赢家要清）
            if = {
                limit = { has_idea = FON_AI_war_fatigue }
                remove_ideas = FON_AI_war_fatigue
            }
            if = {
                limit = { has_country_flag = FON_cw_rear_opened }
                clr_country_flag = FON_cw_rear_opened
            }
            FROM = {
                if = {
                    limit = { has_idea = FON_AI_war_fatigue }
                    remove_ideas = FON_AI_war_fatigue
                }
                if = {
                    limit = { has_country_flag = FON_cw_rear_opened }
                    clr_country_flag = FON_cw_rear_opened
                }
            }
            # === 第一阶段：内部对战（HZH vs FOM / PBF vs FOD）===
            # 不限AI/玩家，投降即被吞并
            if = {
                limit = {
                    ROOT = { tag = HZH }
                    FROM = { tag = FOM }
                }
                FROM = {
                    annex_country = {
                        target = HZH
                        transfer_troops = yes
                    }
                }
            }
            if = {
                limit = {
                    ROOT = { tag = FOM }
                    FROM = { tag = HZH }
                }
                FROM = {
                    annex_country = {
                        target = FOM
                        transfer_troops = yes
                    }
                }
            }
            if = {
                limit = {
                    ROOT = { tag = PBF }
                    FROM = { tag = FOD }
                }
                FROM = {
                    annex_country = {
                        target = PBF
                        transfer_troops = yes
                    }
                }
            }
            if = {
                limit = {
                    ROOT = { tag = FOD }
                    FROM = { tag = PBF }
                }
                FROM = {
                    annex_country = {
                        target = FOD
                        transfer_troops = yes
                    }
                }
            }

            # === 检测两组内战是否都已结束，标记进入最终决战阶段 ===
            if = {
                limit = {
                    # 第一组已决出胜负：HZH和FOM不同时存在
                    NOT = { AND = { HZH = { exists = yes } FOM = { exists = yes } } }
                    # 第二组已决出胜负：PBF和FOD不同时存在
                    NOT = { AND = { PBF = { exists = yes } FOD = { exists = yes } } }
                    # 尚未标记
                    NOT = { has_global_flag = FON_final_war_started }
                }
                set_global_flag = FON_final_war_started
            }

            # === 第二阶段：最终决战投降吞并 ===
            if = {
                limit = { has_global_flag = FON_final_war_started }
                if = {
                    limit = { ROOT = { tag = HZH } }
                    FROM = {
                        annex_country = {
                            target = HZH
                            transfer_troops = yes
                        }
                    }
                }
                if = {
                    limit = { ROOT = { tag = FOM } }
                    FROM = {
                        annex_country = {
                            target = FOM
                            transfer_troops = yes
                        }
                    }
                }
                if = {
                    limit = { ROOT = { tag = PBF } }
                    FROM = {
                        annex_country = {
                            target = PBF
                            transfer_troops = yes
                        }
                    }
                }
                if = {
                    limit = { ROOT = { tag = FOD } }
                    FROM = {
                        annex_country = {
                            target = FOD
                            transfer_troops = yes
                        }
                    }
                }
            # === 最终胜者新闻 ===
            if = {
                limit = {
                    has_global_flag = FON_final_war_started
                    FROM = { tag = FOD }
                    NOT = { FOM = { exists = yes } }
                    NOT = { HZH = { exists = yes } }
                    NOT = { PBF = { exists = yes } }
                }
                FROM = {
                    news_event = { id = FON_civil_war_news.2 }
                }
            }
            if = {
                limit = {
                    has_global_flag = FON_final_war_started
                    FROM = { tag = FOM }
                    NOT = { FOD = { exists = yes } }
                    NOT = { HZH = { exists = yes } }
                    NOT = { PBF = { exists = yes } }
                }
                FROM = {
                    news_event = { id = FON_civil_war_news.3 }
                }
            }
            if = {
                limit = {
                    has_global_flag = FON_final_war_started
                    FROM = { tag = HZH }
                    NOT = { FOD = { exists = yes } }
                    NOT = { FOM = { exists = yes } }
                    NOT = { PBF = { exists = yes } }
                }
                FROM = {
                    news_event = { id = FON_civil_war_news.4 }
                }
            }
            if = {
                limit = {
                    has_global_flag = FON_final_war_started
                    FROM = { tag = PBF }
                    NOT = { FOD = { exists = yes } }
                    NOT = { FOM = { exists = yes } }
                    NOT = { HZH = { exists = yes } }
                }
                FROM = {
                    news_event = { id = FON_civil_war_news.5 }
                }
            }
            }
        }
    }
    # 枫丹内战（两段通用）：任一战区宣战时给 5 天静默窗口
    on_declare_war = {
        effect = {
            if = {
                limit = {
                    OR = {
                        tag = FOD
                        tag = FOM
                        tag = HZH
                        tag = PBF
                    }
                    FROM = {
                        OR = {
                            tag = FOD
                            tag = FOM
                            tag = HZH
                            tag = PBF
                        }
                    }
                }
                set_global_flag = {
                    flag = FON_cw_startup
                    value = 1
                    days = 5
                }
                # 内战厌战：开战 120 天后只给 AI 挂隐藏 idea1
                country_event = { days = 120 id = FON_cw_fatigue.1 }
                FROM = { country_event = { days = 120 id = FON_cw_fatigue.1 } }
                # 300 天后给 AI 挂隐藏 idea2
                country_event = { days = 300 id = FON_cw_fatigue.2 }
                FROM = { country_event = { days = 300 id = FON_cw_fatigue.2 } }



                # FOD vs PBF：200 天后给 AI FOD 开后方登陆场，破地峡僵持
                if = {
                    limit = {
                        OR = {
                            AND = { tag = FOD FROM = { tag = PBF } }
                            AND = { tag = PBF FROM = { tag = FOD } }
                        }
                    }
                    FOD = { country_event = { days = 200 id = FON_cw_rear.1 } }
                }
            }
        }
    }
}
