SUM_Neo_Init = {
	if = {
		limit = { NOT = { has_variable = SUM-Oracle-Index } }
		set_variable = { SUM-Oracle-Index = 20 }
		set_variable = { SUM-Canon-Index = 70 }
		set_variable = { SUM-Genesis-Index = 10 }
	}
	clamp_variable = { var = SUM-Oracle-Index min = 0 max = 100 }
	clamp_variable = { var = SUM-Canon-Index min = 0 max = 100 }
	clamp_variable = { var = SUM-Genesis-Index min = 0 max = 100 }
	clamp_variable = { var = SUM-Genesis-Index min = 0 max = 200 }
	SUM_Neo_Tier_Refresh = yes
}

# ==================== 档位重整 ====================
# 每一档修正只在推进该轴的国策完成时被挂上，旧档同时卸下。
SUM_Neo_Tier_Refresh = {
	remove_dynamic_modifier = { modifier = SUM_Neo_Oracle_T1 }
	remove_dynamic_modifier = { modifier = SUM_Neo_Oracle_T2 }
	remove_dynamic_modifier = { modifier = SUM_Neo_Oracle_T3 }
	remove_dynamic_modifier = { modifier = SUM_Neo_Oracle_T4 }
	remove_dynamic_modifier = { modifier = SUM_Neo_Oracle_T5 }
	remove_dynamic_modifier = { modifier = SUM_Neo_Canon_T1 }
	remove_dynamic_modifier = { modifier = SUM_Neo_Canon_T2 }
	remove_dynamic_modifier = { modifier = SUM_Neo_Canon_T3 }
	remove_dynamic_modifier = { modifier = SUM_Neo_Canon_T4 }
	remove_dynamic_modifier = { modifier = SUM_Neo_Canon_T5 }
	remove_dynamic_modifier = { modifier = SUM_Neo_Combo_Twin_High }
	remove_dynamic_modifier = { modifier = SUM_Neo_Combo_Twin_Low }
	remove_dynamic_modifier = { modifier = SUM_Neo_Combo_A_High }
	remove_dynamic_modifier = { modifier = SUM_Neo_Combo_B_High }
	if = {
		limit = {
			has_variable = SUM-Oracle-Index
			NOT = { check_variable = { SUM-Oracle-Index > 20 } }
		}
		add_dynamic_modifier = { modifier = SUM_Neo_Oracle_T1 }
	}
	if = {
		limit = {
			has_variable = SUM-Oracle-Index
			check_variable = { SUM-Oracle-Index > 20 }
			NOT = { check_variable = { SUM-Oracle-Index > 40 } }
		}
		add_dynamic_modifier = { modifier = SUM_Neo_Oracle_T2 }
	}
	if = {
		limit = {
			has_variable = SUM-Oracle-Index
			check_variable = { SUM-Oracle-Index > 40 }
			NOT = { check_variable = { SUM-Oracle-Index > 60 } }
		}
		add_dynamic_modifier = { modifier = SUM_Neo_Oracle_T3 }
	}
	if = {
		limit = {
			has_variable = SUM-Oracle-Index
			check_variable = { SUM-Oracle-Index > 60 }
			NOT = { check_variable = { SUM-Oracle-Index > 80 } }
		}
		add_dynamic_modifier = { modifier = SUM_Neo_Oracle_T4 }
	}
	if = {
		limit = {
			has_variable = SUM-Oracle-Index
			check_variable = { SUM-Oracle-Index > 80 }
		}
		add_dynamic_modifier = { modifier = SUM_Neo_Oracle_T5 }
	}
	if = {
		limit = {
			has_variable = SUM-Canon-Index
			NOT = { check_variable = { SUM-Canon-Index > 20 } }
		}
		add_dynamic_modifier = { modifier = SUM_Neo_Canon_T1 }
	}
	if = {
		limit = {
			has_variable = SUM-Canon-Index
			check_variable = { SUM-Canon-Index > 20 }
			NOT = { check_variable = { SUM-Canon-Index > 40 } }
		}
		add_dynamic_modifier = { modifier = SUM_Neo_Canon_T2 }
	}
	if = {
		limit = {
			has_variable = SUM-Canon-Index
			check_variable = { SUM-Canon-Index > 40 }
			NOT = { check_variable = { SUM-Canon-Index > 60 } }
		}
		add_dynamic_modifier = { modifier = SUM_Neo_Canon_T3 }
	}
	if = {
		limit = {
			has_variable = SUM-Canon-Index
			check_variable = { SUM-Canon-Index > 60 }
			NOT = { check_variable = { SUM-Canon-Index > 80 } }
		}
		add_dynamic_modifier = { modifier = SUM_Neo_Canon_T4 }
	}
	if = {
		limit = {
			has_variable = SUM-Canon-Index
			check_variable = { SUM-Canon-Index > 80 }
		}
		add_dynamic_modifier = { modifier = SUM_Neo_Canon_T5 }
	}
	if = {
		limit = {
			has_variable = SUM-Oracle-Index
			has_variable = SUM-Canon-Index
			check_variable = { SUM-Oracle-Index > 60 }
			check_variable = { SUM-Canon-Index > 60 }
		}
		add_dynamic_modifier = { modifier = SUM_Neo_Combo_Twin_High }
	}
	if = {
		limit = {
			has_variable = SUM-Oracle-Index
			has_variable = SUM-Canon-Index
			check_variable = { SUM-Oracle-Index < 40 }
			check_variable = { SUM-Canon-Index < 40 }
		}
		add_dynamic_modifier = { modifier = SUM_Neo_Combo_Twin_Low }
	}
	if = {
		limit = {
			has_variable = SUM-Oracle-Index
			has_variable = SUM-Canon-Index
			check_variable = { SUM-Oracle-Index > 60 }
			check_variable = { SUM-Canon-Index < 40 }
		}
		add_dynamic_modifier = { modifier = SUM_Neo_Combo_A_High }
	}
	if = {
		limit = {
			has_variable = SUM-Oracle-Index
			has_variable = SUM-Canon-Index
			check_variable = { SUM-Oracle-Index < 40 }
			check_variable = { SUM-Canon-Index > 60 }
		}
		add_dynamic_modifier = { modifier = SUM_Neo_Combo_B_High }
	}
}
