############################################################
# 璃月内战 · 六方机制挂载（合并文件）
# 钩子：on_startup 初始化；on_monthly 月度保活（原 on_daily 太密，已改为每月一次）
############################################################
on_actions = {
	on_startup = {
		every_country = {
			limit = {
				OR = {
					tag = KQP
					tag = LYY
					tag = GYP
					tag = SHP
					tag = YLH
					tag = CYG
				}
			}
			if = {
				limit = { tag = KQP }
				KQP_CivilWar_Init = yes
			}
			if = {
				limit = { tag = LYY }
				LYY_CivilWar_Init = yes
			}
			if = {
				limit = { tag = GYP }
				GYP_CivilWar_Init = yes
			}
			if = {
				limit = { tag = SHP }
				SHP_CivilWar_Init = yes
			}
			if = {
				limit = { tag = YLH }
				YLH_CivilWar_Init = yes
			}
			if = {
				limit = { tag = CYG }
				CYG_CivilWar_Init = yes
			}
		}
	}
	on_monthly = {
		every_country = {
			limit = {
				OR = {
					tag = KQP
					tag = LYY
					tag = GYP
					tag = SHP
					tag = YLH
					tag = CYG
				}
			}
			if = {
				limit = { tag = KQP }
				KQP_CivilWar_Sync = yes
			}
			if = {
				limit = { tag = LYY }
				LYY_CivilWar_Sync = yes
			}
			if = {
				limit = { tag = GYP }
				GYP_CivilWar_Sync = yes
			}
			if = {
				limit = { tag = SHP }
				SHP_CivilWar_Sync = yes
			}
			if = {
				limit = { tag = YLH }
				YLH_CivilWar_Sync = yes
			}
			if = {
				limit = { tag = CYG }
				CYG_CivilWar_Sync = yes
			}
		}
	}
}
